\section{Introduction}

\begin{frame}{Project Overview}
\begin{itemize}
  \item \textbf{Goal:} Classify human subjects based on brain connectivity networks
  \item \textbf{Dataset:} 114 subjects with 70 brain regions each
  \item \textbf{Three Classification Tasks:}
  \begin{itemize}
    \item Gender (Male vs. Female)
    \item Mathematical Ability (High vs. Normal)
    \item Creativity (Creative vs. Normal)
  \end{itemize}
  \item \textbf{Approach:} Advanced machine learning on graph features
\end{itemize}

\vskip 0.5cm
\small{\textit{Using brain networks to predict cognitive and biological traits}}
\end{frame}

\begin{frame}{Dataset Description}
\begin{columns}
\column{0.5\textwidth}
\textbf{Brain Networks:}
\begin{itemize}
  \item 114 subjects (age 18-29)
  \item 70 brain regions per subject
  \item Weighted connectivity graphs
  \item 2,415 possible connections
  \item Sparse matrices (.mat files)
\end{itemize}

\column{0.5\textwidth}
\textbf{Subject Metadata:}
\begin{itemize}
  \item Gender (50 F, 64 M)
  \item CCI Score (math ability)
  \item CAQ Score (creativity)
  \item FSIQ (IQ), Age
  \item Personality traits
\end{itemize}
\end{columns}

\vskip 0.3cm
\textbf{Classification Tasks:}
\begin{itemize}
  \item \textbf{Gender:} Biological sex classification (50 F, 64 M)
  \item \textbf{Math (75th):} Top 25\% mathematical ability (29 high, 85 normal)
  \item \textbf{Creativity (75th):} Top 25\% creative achievement (31 high, 83 normal)
\end{itemize}
\end{frame}

\section{Methodology}

\begin{frame}{Feature Extraction}
\textbf{Extracted 60+ Graph-Level Features:}

\begin{columns}
\column{0.5\textwidth}
\small{
\textbf{Basic Metrics:}
\begin{itemize}
  \item Degree statistics (avg, max, std)
  \item Strength statistics
  \item Network density
  \item Edge weight statistics
\end{itemize}

\vspace{0.2cm}

\textbf{Clustering \& Topology:}
\begin{itemize}
  \item Clustering coefficient
  \item Transitivity
  \item Small-world properties
  \item Assortativity
\end{itemize}
}

\column{0.5\textwidth}
\small{
\textbf{Centrality Measures:}
\begin{itemize}
  \item Betweenness centrality
  \item Eigenvector centrality
  \item Closeness centrality
\end{itemize}

\vspace{0.2cm}

\textbf{Advanced Features:}
\begin{itemize}
  \item Global/local efficiency
  \item Modularity \& communities
  \item Spectral properties
  \item Rich club coefficient
  \item K-core decomposition
\end{itemize}
}
\end{columns}

\vskip 0.3cm
\centering
\textbf{13 basic features} $\rightarrow$ \textbf{60+ advanced features}
\end{frame}





\begin{frame}{Classification Pipeline}
\begin{enumerate}
  \item \textbf{Feature Extraction:} 60+ features per subject using NetworkX
  \item \textbf{Feature Engineering:} Create interaction terms, ratios, power transforms
  \item \textbf{Feature Selection:} SelectKBest (ANOVA F-test) - choose top 25-35 features
  \item \textbf{Data Split:} 5-fold stratified cross-validation
  \item \textbf{Standardization:} RobustScaler (zero mean, unit variance)
  \item \textbf{Model Training:} 10 algorithms + ensemble methods
  \item \textbf{Evaluation:} Accuracy, Precision, Recall, F1, ROC-AUC
\end{enumerate}

\vskip 0.3cm
\textbf{Key Techniques:}
\begin{itemize}
  \item Hyperparameter optimization (Grid Search)
  \item Class balancing for imbalanced datasets
  \item Stacking ensemble for best performance
\end{itemize}
\end{frame}

\section{Machine Learning Models}

\begin{frame}{Classification Tasks}
\textbf{Three Prediction Tasks:}

\begin{enumerate}
  \item \textbf{Math Ability (Math\_75th):} Predict top 25\% mathematical ability
  \begin{itemize}

  \end{itemize}
  
  \vspace{0.3cm}
  
  \item \textbf{Gender:} Predict biological sex from brain structure
  \begin{itemize}
    \item 64 male vs 50 female subjects
  \end{itemize}

\begin{frame}{Optimization Techniques}
    \begin{enumerate}
        \item \textbf{Feature Engineering:}
        \begin{itemize}
            \item Interaction terms (multiplicative combinations)
            \item Ratio features (strength dominance, degree variability)
            \item Power transformations (squared, square root)
            \item Statistical aggregates (sum, mean, std, range)
        \end{itemize}
        
        \item \textbf{Hyperparameter Optimization:}
        \begin{itemize}
            \item Grid search with 5-fold cross-validation
            \item Task-specific parameter tuning
            \item Class balancing for imbalanced datasets
        \end{itemize}
        
        \item \textbf{Regularization:}
        \begin{itemize}
            \item PCA dimensionality reduction (prevents overfitting)
            \item Robust scaling (handles outliers)
            \item Feature selection (SelectKBest, ANOVA F-test)
        \end{itemize}
        
        \item \textbf{Validation:}
        \begin{itemize}
            \item Stratified 5-fold cross-validation
            \item Multiple random seeds for stability
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}{Understanding the Visualizations}
\textbf{All metrics computed using 5-fold cross-validation:}

\begin{enumerate}
  \item \textbf{Confusion Matrix:} Shows prediction errors
  \begin{itemize}
    \item Diagonal = Correct predictions
    \item Off-diagonal = Misclassifications
    \item Based on CV predictions (not a separate test set)
  \end{itemize}
  
  \vspace{0.2cm}
  
  \item \textbf{ROC Curve:} Measures discrimination ability
  \begin{itemize}
    \item AUC = Area Under Curve (0.5 = random, 1.0 = perfect)
    \item Shows tradeoff between true/false positive rates
  \end{itemize}
  
  \vspace{0.2cm}
  
  \item \textbf{Learning Curve:} Shows if model needs more data
  \begin{itemize}
    \item Training vs validation performance
    \item Gap indicates overfitting
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Validation Methodology}
\begin{alertblock}{Cross-Validation vs Train/Test Split}
\textbf{Our Approach:} 5-fold stratified cross-validation
\begin{itemize}
  \item Dataset split into 5 equal parts
  \item Each part serves as validation set once
  \item Model trained on remaining 4 parts
  \item All 114 subjects used for both training and validation
\end{itemize}
\end{alertblock}

\vspace{0.3cm}

\textbf{Why Cross-Validation?}
\begin{enumerate}
  \item \textbf{Small Dataset:} 114 subjects total
  \begin{itemize}
    \item Math: only 29 positive cases
    \item Train/test split would lose valuable data
  \end{itemize}
  
  \item \textbf{Robust Estimates:} Each subject predicted exactly once
  
  \item \textbf{Standard Practice:} Used in medical/neuroscience research
\end{enumerate}

\vspace{0.2cm}

\end{frame}
